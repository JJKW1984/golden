**Goal:** Implement debt account management, payment ledger integration, monthly amortization with exact integer-cents rounding, projection scenarios, and "Update balance" adjustment flow.

**Architecture:** Debt payments flow through the ledger service as transactions with `link_type='debt'` and principal/interest split. The `services/debt_payoff.py` module computes monthly amortization using integer-cents arithmetic, handles the "payment ≤ interest" edge case, and generates projection scenarios (+$0, +$50, +$100 over current allocation). The debt router wires HTTP endpoints to ledger and projection services. The debt screen displays the active target (balance, % paid off, minimum, extra this month, payoff date) and debt queue sorted by snowball/avalanche method. All derived balances remain recomputable via `recompute_balances(ctx)`.

**Tech Stack:** Python 3.11+, FastAPI, SQLAlchemy ORM, HTMX, Chart.js (CDN), Tailwind, Pydantic v2, pytest

```python
# finapp/services/debt_payoff.py
from decimal import Decimal, ROUND_HALF_UP

def amortize_month(opening_cents: int, interest_rate_bps: int, payment_cents: int) -> tuple[int, int]:
    """
    Compute principal and interest portions of a debt payment in a single month.
    
    Args:
        opening_cents: opening balance in cents (INTEGER)
        interest_rate_bps: annual interest rate in basis points (INTEGER)
        payment_cents: amount paid this month in cents (INTEGER)
    
    Returns:
        (principal_cents, interest_cents) tuple, both INTEGER cents
        
    Rules:
    - interest = opening * (interest_rate_bps / 10000) / 12, rounded half-up to nearest cent
    - principal = max(payment - interest, 0) — never negative
    - If principal = 0, payment went entirely to interest (edge case flagged)
    """
    # interest = opening_cents * (interest_rate_bps / 10000 / 12)
    # Using Decimal to avoid float:
    # interest_cents = opening_cents * interest_rate_bps / 10000 / 12
    interest_cents = int(
        (Decimal(opening_cents) * Decimal(interest_rate_bps) / Decimal(10000) / Decimal(12)).quantize(
            Decimal('1'), rounding=ROUND_HALF_UP
        )
    )
    
    principal_cents = max(payment_cents - interest_cents, 0)
    
    return (principal_cents, interest_cents)
```

```python
# finapp/services/debt_payoff.py
from datetime import datetime, timedelta

def project_debt_payoff(
    opening_balance_cents: int,
    interest_rate_bps: int,
    monthly_payment_cents: int,
    scenario: str = 'minimum'
) -> tuple[int, int, str, bool]:
    """
    Project debt payoff timeline under a fixed monthly payment.
    
    Args:
        opening_balance_cents: opening balance in cents
        interest_rate_bps: annual interest rate in basis points
        monthly_payment_cents: fixed monthly payment in cents
        scenario: 'minimum', 'current', '+50', '+100' (label only, all use monthly_payment_cents)
    
    Returns:
        (months_to_payoff, total_interest_cents, payoff_date_str, payment_gte_interest_flag)
        - payoff_date_str: "YYYY-MM-DD" or "Never" if payment ≤ interest every month
        - payment_gte_interest_flag: True if payment always covers interest, False if ever (payment ≤ interest)
    """
    balance = opening_balance_cents
    total_interest = 0
    months = 0
    payment_always_gte_interest = True
    today = datetime.now()
    
    max_months = 1200  # 100 years failsafe
    
    while balance > 0 and months < max_months:
        principal, interest = amortize_month(balance, interest_rate_bps, monthly_payment_cents)
        
        if principal == 0:
            payment_always_gte_interest = False
        
        balance -= principal
        total_interest += interest
        months += 1
    
    if balance > 0:
        payoff_date_str = "Never"
    else:
        payoff_date = today + timedelta(days=30 * months)
        payoff_date_str = payoff_date.strftime("%Y-%m-%d")
    
    return (months, total_interest, payoff_date_str, payment_always_gte_interest)
```

```python
# finapp/services/ledger.py

def create_debt_adjustment(
    ctx: AccountContext,
    debt_id: int,
    adjustment_cents: int,
    statement_balance_cents: int
) -> Transaction:
    """
    Create a debt adjustment transaction when actual balance differs from cached.
    
    Args:
        adjustment_cents: negative if balance was overstated, positive if understated
        statement_balance_cents: the correct balance per statement
    """
    debt = session.query(DebtAccount).filter_by(id=debt_id, account_id=ctx.account_id).first()
    if not debt:
        raise ValueError(f"Debt account {debt_id} not found")
    
    direction = "in" if adjustment_cents > 0 else "out"
    
    txn = Transaction(
        account_id=ctx.account_id,
        date=datetime.now().date(),
        kind="adjustment",  # or "debt_adjustment"
        category_id=None,
        amount_cents=abs(adjustment_cents),
        direction=direction,
        payee="",
        mood_tag=None,
        memo=f"Update balance to match your statement: ${statement_balance_cents/100:.2f}",
        principal_cents=adjustment_cents,
        interest_cents=0,
        link_type="debt",
        link_id=debt_id,
        is_deleted=False
    )
    session.add(txn)
    session.commit()
    
    return txn
```

```python
    ctx: AccountContext = Depends(get_account_context)

    session = SessionLocal()
    debt = session.query(DebtAccount).filter_by(id=debt_id, account_id=ctx.account_id).first()

        return {"error": "Debt not found"}, 404

    from finapp.services.ledger import create_transaction

        date_str=datetime.now().isoformat().split('T')[0],
        amount_cents=request.payment_cents,
        direction="out",
        category_name="Debt",
        payee=debt.name,
        mood_tag=None,
        memo=request.memo or f"Payment to {debt.name}",
        link_type="debt",
        link_id=debt_id
    )
    
    # Update cached balance (principal only)
    debt.cached_balance_cents -= txn.principal_cents
    session.commit()
    
    return {
        "transaction_id": txn.id,
        "principal": txn.principal_cents / 100,
        "interest": txn.interest_cents / 100,
        "new_balance": debt.cached_balance_cents / 100
    }
```

```python
@router.get("/{debt_id}/projection", response_class=HTMLResponse)
def get_debt_projection_view(
    debt_id: int,
    ctx: AccountContext = Depends(get_account_context)
):
    """Render projection chart for a debt."""
    session = SessionLocal()
    debt = session.query(DebtAccount).filter_by(id=debt_id, account_id=ctx.account_id).first()
    
    if not debt:
        return {"error": "Debt not found"}, 404
    
    from finapp.services.debt_payoff import project_debt_payoff
    
    minimum = debt.minimum_payment_cents or 0
    current = debt.monthly_allocation_cents or minimum
    
    # Generate scenarios with monthly balances for charting
    scenarios = []
    colors = ['#ef4444', '#f97316', '#22c55e']  # red, orange, green
    
    for i, (name, payment, color) in enumerate([
        ('minimum', minimum, colors[0]),
        ('+$50', current + 50, colors[1]),
        ('+$100', current + 100, colors[2])
    ]):
        months, total_interest, payoff_date, payment_gte = project_debt_payoff(
            debt.cached_balance_cents,
            debt.interest_rate_bps,
            payment
        )
        
        # Generate month-by-month balances for the chart
        balance = debt.cached_balance_cents
        monthly_balances = [balance]
        
        for month in range(months):
            principal, interest = amortize_month(balance, debt.interest_rate_bps, payment)
            balance -= principal
            monthly_balances.append(max(0, balance))
        
        scenarios.append({
            "name": name,
            "months": months,
            "total_interest": total_interest,
            "payoff_date": payoff_date,
            "monthly_balances": monthly_balances,
            "color": color,
            "payment_gte": payment_gte
        })
    
    env = Environment(loader=FileSystemLoader("finapp/templates"))
    template = env.get_template("debt/projection.html")
    
    import json
    
    return template.render(
        debt={"name": debt.name, "id": debt.id},
        scenarios=scenarios,
        scenarios_json=json.dumps(scenarios)
    )
```

```python
from pydantic import BaseModel

class DebtAdjustRequest(BaseModel):
    statement_balance_cents: int

@router.post("/{debt_id}/adjust")
def post_debt_adjust(
    debt_id: int,
    request: DebtAdjustRequest,
    ctx: AccountContext = Depends(get_account_context)
):
    """Adjust debt balance to match statement."""
    session = SessionLocal()
    debt = session.query(DebtAccount).filter_by(id=debt_id, account_id=ctx.account_id).first()
    
    if not debt:
        return {"error": "Debt not found"}, 404
    
    from finapp.services.ledger import create_debt_adjustment
    
    # Create adjustment transaction
    adjustment = request.statement_balance_cents - debt.cached_balance_cents
    
    txn = create_debt_adjustment(
        ctx=ctx,
        debt_id=debt_id,
        adjustment_cents=adjustment,
        statement_balance_cents=request.statement_balance_cents
    )
    
    # Update cached balance
    debt.cached_balance_cents = request.statement_balance_cents
    session.commit()
    
    return {
        "transaction_id": txn.id,
        "adjustment": adjustment / 100,
        "new_balance": debt.cached_balance_cents / 100
    }
```
